\section*{Publications}

\publication{\textbf{Li, Y.}, \& Cao, J. (2026). Diagnosing the increase in East Asian summer monsoon precipitation under a low-emissions scenario from an energy perspective (\textit{Manuscript in Preparation}).}

\researchsummary{Using multiple climate model simulations together with moisture budget and moist static energy budget analyses, we diagnose the increase in East Asian summer precipitation under the SSP1-2.6 scenario relative to other scenarios. Our results suggest that reduced anthropogenic aerosols over East Asia increase surface solar radiation and evaporation, thereby enhancing convective activity and large-scale upward moisture transport, intensifying the East Asian summer monsoon circulation, and increasing rainfall.}

\publication{\textbf{Li, Y.}, Lin, M., Naik, V., \& Horowitz, L. (2026). Recent tropospheric ozone trends in the northern mid-latitudes: roles of Asian emission reductions and climate variability (\textit{Manuscript in Preparation}).}

\researchsummary{Using high-elevation observations and the GFDL climate model, we show that reductions in Asian emissions are the primary driver of the decreasing free-tropospheric ozone trend over the western US. Stratosphere-troposphere transport (STT) strongly modulates the interannual variability of free-tropospheric ozone over the western US. Climate variability modulates ozone anomalies over the western US by influencing STT through changes in the meandering of the polar jet, rather than the Pacific subtropical jet.}

\publication{\textbf{Li, Y.}, \& Shen, X. (2026). Using causal inference to quantify the roles of tropospheric wave sources and stratospheric preconditioning in sudden stratospheric warmings (\textit{Manuscript in Preparation}).}

\researchsummary{Apply a physically based causal inference framework to quantify the relative roles of tropospheric wave sources and stratospheric preconditioning in influencing the occurrence of sudden stratospheric warmings (SSWs). Our results indicate that both factors play essential roles in the occurrence of SSWs. Approximately one-third of the effect of tropospheric wave sources and one-half of the effect of stratospheric preconditioning on stratospheric polar vortex variability are mediated by stratospheric wave-activity fluxes.}
